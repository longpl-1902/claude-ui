## Goal
Create the electron-vite + React 18 + TypeScript strict project, with every Phase 1 dependency declared and installed. Add a secure BrowserWindow, a typed preload bridge, the shared domain types and the IPC channel contract. All later tasks only add or replace source files and never touch package.json, the lockfile or the configs. The task is larger than usual because it is the single owner of all tooling.

## Notes
**Do not scaffold with an interactive `npm create`.** Write the files by hand. Before pinning versions, check compatibility with `npm view <pkg> version` / `peerDependencies`, then pin exact versions with no `^`.

**Local desktop app only (R-18).** There is no web build target and no hosting config. Do not reference any CDN or remote URL anywhere.

### src/shared/types.ts
Export exactly these names. Later tasks depend on them.
```ts
export type AgentEventKind = 'tool_start' | 'tool_end' | 'message' | 'status' | 'error' | 'spawn' | 'stop';
export type AgentEvent = {            // verbatim from spec.md
  ts: string; sessionId: string; agentId: string; parentAgentId?: string;
  kind: AgentEventKind;
  tool?: string; target?: string;
  result?: string; tokens?: { input: number; output: number };
  worktree?: string; accountId: string;
};
export type AgentStatus = 'running' | 'waiting' | 'needs-you' | 'done';
export type ActivityKind = 'think' | 'read' | 'edit' | 'bash' | 'wait' | 'error';
export interface TimelineSegment { kind: ActivityKind; startPct: number; widthPct: number }
export interface ActivityItem { time: string; tool: string; target: string; result: string; failed?: boolean }
export interface ChangedFile { path: string; added: number; removed: number }
export interface MailboxMessage { direction: 'in' | 'out'; peer: string; time: string; text: string }
export interface Agent {
  id: string; name: string; initials: string; avatarColor: string; role: string; model: string;
  worktree: string;              // branch name, e.g. 'feat/api-schema'
  taskTitle: string;             // '#2 Schema v2 for SheetDto'
  current: { tool: string; target: string };
  elapsedSec: number; tokens: number; toolCalls: number;
  timeline: TimelineSegment[]; activity: ActivityItem[]; files: ChangedFile[]; messages: MailboxMessage[];
}
export interface Worktree { branch: string; path: string; agentIds: string[]; changedFiles: number; ahead: number; overlapsWith?: string[] }
export type SharedTaskState = 'done' | 'in_progress' | 'blocked' | 'failing';
export interface SharedTask { id: number; title: string; state: SharedTaskState; ownerAgentId: string; blockedBy?: number; failingCount?: number }
export interface FileOverlap { path: string; agentIds: string[]; worktrees: string[] }
export interface ConsoleLine { time: string; who: string; text: string }   // who = 'you' or an agent id
export interface TeamSnapshot {
  repo: string; teamName: string; accountId: string; now: string;   // ISO timestamp used as "now" for derivation
  agents: Agent[]; events: AgentEvent[]; worktrees: Worktree[]; tasks: SharedTask[];
  overlaps: FileOverlap[]; console: ConsoleLine[]; timelineTicks: string[];
}
export interface FolderRule { pattern: string; accountId: string | 'ask'; note: string }
export interface AccountProfile { id: string; name: string; color: string; configDir: string; folderRules: FolderRule[] }  // spec
export interface AccountView extends AccountProfile { email?: string; org?: string; plan?: string; signedIn: boolean; tokensToday?: number }
export interface DetectedLogin { configDir: string; email?: string; org?: string; sources: string[]; linkedAccountId?: string; signedIn: boolean; suggestedAccountId?: string }
export type LimitPolicy = 'pause' | 'wait' | 'switch';
export interface SafetySettings { warnPersonalInCompanyFolder: boolean; showAccountColorEdge: boolean; lockAccountOnTeamStart: boolean }
export interface RunningSession { teamName: string; path: string; accountId: string; agentCount: number }
export interface AccountsSnapshot {
  accounts: AccountView[]; defaultAccountId: string; activeAccountId: string;
  detected: DetectedLogin[]; rules: FolderRule[]; safety: SafetySettings; limitPolicy: LimitPolicy; sessions: RunningSession[];
}
export type SignInStep = 'none' | 'waiting' | 'done';
export interface OkResult { ok: boolean; reason?: string }
```

### src/shared/ipc.ts
```ts
export const IPC = {
  teamGetSnapshot: 'team:get-snapshot',
  teamEvent: 'team:event',                 // main -> renderer push (unused in Phase 1)
  accountsGetSnapshot: 'accounts:get-snapshot',
  accountsSetDefault: 'accounts:set-default',
  accountsStartSignIn: 'accounts:start-sign-in',
  accountsCheckSignIn: 'accounts:check-sign-in',
  accountsCancelSignIn: 'accounts:cancel-sign-in',
  agentSendMessage: 'agent:send-message',
  agentPause: 'agent:pause',
  agentStop: 'agent:stop',
  teamPauseAll: 'team:pause-all',
  worktreeOpenInEditor: 'worktree:open-in-editor',
  appGetVersion: 'app:get-version',
} as const;
export type IpcChannel = typeof IPC[keyof typeof IPC];
export interface IpcInvokeMap {      // args tuple + result, one entry per invoke channel (all except teamEvent)
  'team:get-snapshot': { args: []; result: TeamSnapshot };
  'accounts:get-snapshot': { args: []; result: AccountsSnapshot };
  'accounts:set-default': { args: [accountId: string]; result: OkResult };
  'accounts:start-sign-in': { args: [configDir: string]; result: OkResult };
  'accounts:check-sign-in': { args: [configDir: string]; result: { step: SignInStep; email?: string } };
  'accounts:cancel-sign-in': { args: [configDir: string]; result: OkResult };
  'agent:send-message': { args: [agentId: string, text: string]; result: OkResult };
  'agent:pause': { args: [agentId: string]; result: OkResult };
  'agent:stop': { args: [agentId: string]; result: OkResult };
  'team:pause-all': { args: []; result: OkResult };
  'worktree:open-in-editor': { args: [path: string]; result: OkResult };
  'app:get-version': { args: []; result: string };
}
export type InvokeChannel = keyof IpcInvokeMap;
export const INVOKE_CHANNELS: InvokeChannel[] = [/* every key above */];
export interface ConsoleApi {
  team: { getSnapshot(): Promise<TeamSnapshot>; pauseAll(): Promise<OkResult>; onEvent(cb: (e: AgentEvent) => void): () => void };
  accounts: { getSnapshot(): Promise<AccountsSnapshot>; setDefault(id: string): Promise<OkResult>;
              startSignIn(dir: string): Promise<OkResult>; checkSignIn(dir: string): Promise<{ step: SignInStep; email?: string }>; cancelSignIn(dir: string): Promise<OkResult> };
  agent: { sendMessage(id: string, text: string): Promise<OkResult>; pause(id: string): Promise<OkResult>; stop(id: string): Promise<OkResult> };
  worktree: { openInEditor(path: string): Promise<OkResult> };
  app: { getVersion(): Promise<string> };
}
```
- Add a compile-time check that `INVOKE_CHANNELS` covers every key, for example `satisfies` plus a type-level exhaustiveness helper.
- src/preload/index.ts builds `api: ConsoleApi` with a typed helper `invoke<C extends InvokeChannel>(c: C, ...args: IpcInvokeMap[C]['args']): Promise<IpcInvokeMap[C]['result']>`. `onEvent` uses `ipcRenderer.on(IPC.teamEvent, ...)` and returns an unsubscribe function.
- Import from `@shared/ipc` only. The bundler inlines it, and the preload output must not `require` local files.
- src/preload/index.d.ts: `declare global { interface Window { api?: ConsoleApi } }`. The type is optional so renderer code guards it and tests run without it.
